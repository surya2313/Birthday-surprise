        // Web Audio Synthesizer Music
        let audioCtx;
        let isAudioPlaying = false;
        let musicInterval;

        function playSynthesizedMelody() {
            if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            if (audioCtx.state === 'suspended') audioCtx.resume();

            const notes = [261.63, 261.63, 293.66, 261.63, 349.23, 329.63, 261.63, 261.63, 293.66, 261.63, 392.00, 349.23];
            let noteIdx = 0;

            musicInterval = setInterval(() => {
                if (!isAudioPlaying) return;
                let osc = audioCtx.createOscillator();
                let gain = audioCtx.createGain();
                osc.type = 'sine';
                osc.frequency.setValueAtTime(notes[noteIdx], audioCtx.currentTime);
                gain.gain.setValueAtTime(0.08, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + 0.4);
                osc.connect(gain);
                gain.connect(audioCtx.destination);
                osc.start();
                osc.stop(audioCtx.currentTime + 0.45);
                noteIdx = (noteIdx + 1) % notes.length;
            }, 500);
        }

            if (isAudioPlaying) {

            } else {
                isAudioPlaying = true;
                playSynthesizedMelody();
                document.getElementById('musicBtn').innerText = '🎶';
            }
        }

        // Scratch Card Logic with live progress + auto-reveal celebration
        let scratchInitialized = false;
        let scratchRevealedFired = false;
        function initScratchCard() {
            if (scratchInitialized) return;
            scratchInitialized = true;
            const canvas = document.getElementById('scratchCanvas');
            const ctx = canvas.getContext('2d');

            const grad = ctx.createLinearGradient(0, 0, canvas.width, canvas.height);
            grad.addColorStop(0, '#ff9db3');
            grad.addColorStop(1, '#e84c6b');
            ctx.fillStyle = grad;
            ctx.fillRect(0, 0, canvas.width, canvas.height);
            ctx.fillStyle = '#ffffff';
            ctx.font = 'bold 16px Quicksand, sans-serif';
            ctx.textAlign = 'center';
            ctx.fillText('Scratch with your finger! 🐾', canvas.width / 2, canvas.height / 2 + 5);

            let isScratching = false;
            let lastCheck = 0;

            function scratch(e) {
                if (!isScratching) return;
                const rect = canvas.getBoundingClientRect();
                const x = (e.clientX || (e.touches && e.touches[0].clientX)) - rect.left;
                const y = (e.clientY || (e.touches && e.touches[0].clientY)) - rect.top;
                ctx.globalCompositeOperation = 'destination-out';
                ctx.beginPath();
                ctx.arc(x, y, 22, 0, Math.PI * 2);
                ctx.fill();

                const now = Date.now();
                if (now - lastCheck > 150) {
                    lastCheck = now;
                    checkScratchProgress(ctx, canvas);
                }
            }

            function checkScratchProgress(ctx, canvas) {
                const data = ctx.getImageData(0, 0, canvas.width, canvas.height).data;
                let cleared = 0;
                for (let i = 3; i < data.length; i += 4 * 20) {
                    if (data[i] === 0) cleared++;
                }
                const total = data.length / (4 * 20);
                const pct = Math.min(100, Math.round((cleared / total) * 100));
                document.getElementById('scratchProgress').innerText = pct >= 100 ? "Fully revealed! 🎉" : `Keep scratching... ${pct}% revealed`;
                if (pct >= 65 && !scratchRevealedFired) {
                    scratchRevealedFired = true;
                    document.getElementById('scratchContainer').classList.add('revealed');
                    triggerEmojiExplosion(['💖', '✨', '❤️']);
                }
            }

            canvas.addEventListener('mousedown', () => isScratching = true);
            canvas.addEventListener('mouseup', () => isScratching = false);
            canvas.addEventListener('mousemove', scratch);
            canvas.addEventListener('touchstart', () => isScratching = true);
            canvas.addEventListener('touchend', () => isScratching = false);
            canvas.addEventListener('touchmove', scratch);
        }
